# ============================================================
# requirements.txt — 宠物店智慧经营中枢
#     pip install -r requirements.txt
#
# ⚠️ 需要 Python >= 3.10，这是硬门槛，装不上依赖多半是 Python 太老。
#    其中的决定因素是 langgraph（它自己要求 >=3.10）；pydantic 2 要求 >=3.8。
#    先跑 `python3 --version` 确认。CentOS 7 / 阿里云 ECS 自带的 python3 是 3.6.8，
#    在那个版本上 pip 只会列到 fastapi 0.83.0 —— 那不是网络问题，是 pip 按本机
#    Python 版本过滤的结果。解决办法是装个新的 Python（Miniconda 或 pyenv），
#    不要去改下面的版本号，改到 0.83.0 也装不起来 pydantic 和 langgraph。
#
# 下面的下限取的是「本项目用到的那个 API 最早出现的版本」，不是「开发机上的版本」。
# 括号里是实测跑通的版本，只作参考，不影响安装。
# ============================================================

# ---------- Web 框架 ----------
fastapi>=0.100.0            # 0.100 起才支持 pydantic v2（实测 0.139.0）
uvicorn[standard]>=0.23.0   # [standard] 带上 watchfiles，start.py 的 reload=True 才生效
pydantic>=2.0               # 代码用的是 v2 API：Field(gt=0)、model_dump()

# ---------- 数据库 ----------
PyMySQL>=1.0.0              # 直连 MySQL，没有 ORM（取舍见 README「数据模型」）

# ---------- 配置 ----------
python-dotenv>=0.19.0

# ---------- 向量检索 ----------
numpy>=1.24.0               # 必装：本地文件降级模式用它做余弦暴力检索
qdrant-client>=1.2.0        # 可选：本地嵌入式模式 QC(path=...) 从 1.2 起有。
                            # 装不上会自动降级到本地文件，不影响启动（实测 1.18.0）

# ---------- 工作流 ----------
langgraph>=0.2.0            # 必装：routes.py 是模块级 import，缺了服务起不来。
                            # 只用到 StateGraph / add_node / add_edge / END / compile / invoke，
                            # 这套 API 从 0.1 起就稳定（实测 1.2.0）

# ---------- LLM ----------
openai>=1.0.0               # 可选：用的是 v1 SDK（OpenAI(base_url=...) + chat.completions）。
                            # 没装或 key 无效都会自动降级到规则引擎（实测 2.35.1）
